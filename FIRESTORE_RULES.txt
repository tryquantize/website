rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow anyone to write partner requests
    match /partnerRequests/{document} {
      allow read, write: if true;
    }
    
    // Allow anyone to write company outreach requests
    match /companyOutreachRequests/{document} {
      allow read, write: if true;
    }
    
    // Allow read access to companies for analytics and search
    // Allow write access for analytics counters and admin updates
    match /companies/{document} {
      allow read: if true;
      allow write: if true; // Consider restricting to admin users only in production
    }
    
    // Allow authenticated users to access their own data
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Deny all other access
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
